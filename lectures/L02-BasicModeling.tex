\begin{frame}{Sudoku takeaways}
	\begin{itemize}
		\item The Sudoku problem was formulated in a \defined{declarative} manner: we specify \stressed{what} to solve, not \stressed{how} to solve
		\item High-level modeling languages support high-level constraints such as $\cons{AllDifferent}{\ldots}$ (and more such as $\cons{NoOverlap}{\ldots}$, $\cons{Element}{\ldots}$, $\cons{Min}{\ldots}$, \ldots)
		\item The modeling language provides unified access to vastly different \defined{solver technologies}, each with their own performance characteristics and capabilities
		\item Much engineering and research goes into making the modeling language \stressed{expressive} and the solver \stressed{fast}
	\end{itemize}
\end{frame}

\subsection{Decision variables, domains, and constraints}

\begin{frame}{Decision variables}
	\defined{Decision variables} are the \stressed{unknowns} of the problem.
 % (\eg $st$ denotes whether we try a Stella)
	Each decision variable has a \defined{domain} (\ie a set of values).

	\parBreak

The two primary types of variables are:

  \begin{itemize}
	  \item Boolean variables, \eg $\varSt \in \{0,1\}$ (whether we try a Stella or not)
	  \item Integer variables, \eg $\varSt \in \interval{0}{2}$ (whether we try 0, 1, or 2 Stella's)
  \end{itemize}

	\parBreak
	An \defined{assignment} maps each variable to a value from its domain (\eg $\varSt = 1, \varDu = 0, \ldots$)

  \parBreak
	The minimum value of the domain is called the \defined{lower  bound}, the maximum value the \defined{upper bound}

\parBreak
	In this course, we only consider  domains which are \defined{finite} (no unbounded, \eg $\varSt \in \interval{0}{}$) and \defined{discrete} (no floating point, \eg $\varSt \in \interval{0.0}{2.5}$). Different modeling languages support different variable types (\eg floats, sets, strings, bitvectors, ...)

\end{frame}

\begin{frame}{Expressions, constraints, and solutions}
	A model contains \defined{expressions}, \ie a tree of mathematical symbols. 
	A constraint is a \defined{relation} over variables, \ie an expression which evaluates to \true or \false.
	For a feasible assignment (\ie \defined{solution}), all constraints evaluate to \true.
	% (variables, operators, numbers, \stressed{other expressions}, \ldots).

\begin{frame}{Logical constraints}

	\defined{Logical constraints} involve \defined{Boolean operators} over Boolean-valued expressions

  \begin{columns}
    \begin{column}{0.75\textwidth}
  \begin{description}
	  \item[Negation\demph{, not}:] $\neg \bvA$
	    % \begin{itemize}
	    %  \item \Eg Adding $\neg \varSt$ to the model removes all assignments where $\varSt = 1$
	    % \end{itemize}
	  \item[Or\demph{, disjunction}:] $\bvA \vee \bvB$
	  \item[And\demph{, conjunction}:] $\bvA \wedge \bvB$
	  \item[Implication\demph{, conditional}:] $\bvA \implies \bvB$
	  \item[Equivalence\demph{, bi-conditional, if-and-only-if}:] $\bvA \iff \bvB$, or $\bvA == \bvB$
	  \item[Exclusive-or\demph{, xor}:] $\bvA \xor \bvB$ 
  \end{description}
    \end{column}
    \begin{column}{0.2\textwidth}
	    \centering

  \end{columns}

  \parBreak
  \centering

  \begin{itemize}
	  \item \Eg \qquoted{We're out of Kriek!}: \quad $\neg \varKl$
		  \begin{itemize}
			  \item Disallows the old optimal solution, \texttt{Duvel, Karmeliet, Kriek (232)}

\end{frame}

\begin{frame}{When does a logical constraint evaluate to \true?}
  Each operator is defined by its \defined{truth table}

  % TODO consider colours for T/F

  \parBreak
  \begin{center}
  	
{%\footnotesize
  \begin{tabular}{|c|c||c|c|c|c|c|c|}
    \hline
	  $\bvA$ & $\bvB$ & $\neg \bvB$ & $\bvA \lor \bvB$ & $\bvA \land \bvB$ & $\bvA \iff \bvB$ & $\bvA \xor \bvB$ & $\bvA \implies \bvB$ \\
    \hline
	  F & F & T & F  & F & T & F & {\onslide<2->{T}} \\
	  F & T & F & T  & F & F & T & {\onslide<3->{T}} \\
	  T & F & \ldots & T  & F & F & T & {\onslide<4->{F}} \\
	  T & T & \ldots & T  & T & T & F & {\onslide<5->{T}} \\
    \hline
  \end{tabular}
  }

  \end{center}

  \onslide<6->
  \parBreak
	Boolean-valued expressions are equivalent if they have the same truth table. We can rewrite equivalent expressions (using $\equiv$ notation):

	% \begin{align}
	%     		\bvA \implies \bvB & \equiv \neg \bvA \lor \bvB \\
	% 	\bvA \iff \bvB & \equiv  (\bvA \implies \bvB) \land (\bvB \implies \bvA) \\
	% 	\bvA \xor \bvB & \equiv \bvA \iff \neg \bvB \equiv \bvA \neq \bvB
	% \end{align}

	\begin{itemize}
		\item $\bvA \implies \bvB \equiv \neg \bvA \lor \bvB$
		\item $\bvA \iff \bvB \equiv  (\bvA \implies \bvB) \land (\bvB \implies \bvA)$
		\item $\bvA \xor \bvB \equiv \bvA \iff \neg \bvB \equiv \bvA \neq \bvB$
	\end{itemize}

	% Note the use of $\equiv$ instead of $\iff$ when rewriting equivalent Boolean-valued expression
\end{frame}

\subsection{Integer constraints}

\begin{frame}{Simple comparison constraints}

	Simple comparisons ($=, \neq, <, \leq, >, \geq)$ over an integer variable, \eg $x \in \interval{1}{5}$:
  \begin{align}
	  % TODO perhaps make set of sets or something
	  x &= 3 & \onslide<2->{1}~\text{solution(s)} & \quad & \onslide<2->{\set{x = 3}}\\
	  x & < 4 & \onslide<3->{3}~\text{solution(s)} && \onslide<3->{\set{x = 1, x = 2, x = 3}}\\
	  x &\neq 6 & \onslide<4->{5}~\text{solution(s)} && \onslide<4->{\set{x = 1, \ldots, x = 5}}\\
	  x & = 7 & \onslide<5->{0}~\text{solution(s)} && \onslide<5->{\EmptySet} \\
  \end{align}

  \onslide<6->
	% If $x, y, z \in \interval{1}{5}$, then how many solutions does each constraint leave?
  Note that while these expressions involve integer variables, they still evaluate to Booleans. So, as Boolean-valued expressions, they can be used with Boolean operators:
  \begin{align}
      &(x = 3) \wedge (y > 5) \\
      &(y \neq 6) \rightarrow (a \vee (z < 2))
  \end{align}
	Tip: good use of brackets $(\ldots)$ avoids ambiguity.
\end{frame}

\begin{frame}{Arithmetic constraints}

	Arithmetic constraints combine arithmetic operations $(+,-,*,/)$ over integers with a comparisons ($=, \neq, <, \leq, >, \geq)$, \eg for $x, y \in \interval{1}{5}$:

  \begin{align}
	  x + y &= 3\\
	  x &< y & \\
    y - z &\neq 6 \\
    2*z - (x*y + y) & < 2 \\
  \end{align}

  \pause
  \parBreak

	Integer division $x/y = 5$ is tricky because it is \defined{undefined} in assignments where $y=0$.
Division is a \defined{partial function}.
	Some languages/systems simply forbid a division where the denominator has $0$ in its domain. There are more peculiarities with integer division, such as using \textit{floor division} or \textit{rounding division} for negative numbers.
\end{frame}

\begin{frame}{Integer linear constraints}

	% A \defined{linear expression} is a sum of $n$ \defined{linear terms}: $a_1 \times x_1 + \ldots + a_n  \times x_n$.
	% \parBreak
	% A \defined{linear constraint}

	A \defined{linear term} is a multiplication of a variable with a constant, \eg $2 \times x$.
	A \defined{linear expression} is a sum of linear terms and a constant, \eg $2 \times x - 3 \times y - 6$
	\parBreak

	A \defined{linear constraint} is a comparison between two linear expressions, \eg $2 \times x - 3 \times y - 6 \leq 5 \times z$.
	A \defined{linear \stressed{in}equality} uses only the comparisons $(<, \leq, >, \geq)$.

	\parBreak
	 Linear inequalities can be rewritten to a normal form with integers $a_i, k$ and integer variables $x_i$:

	$$
	a_1 \times x_1 + \cdots + a_n \times x_n \leq k \quad (\eg~2 \times x - 3 \times y - z \leq 6)
	$$

	\parBreak
	We can also normalize \defined{linear equality} and \defined{\stressed{dis}equality constraints}:

	\begin{itemize}
		\item $x + y = 5 \equiv (x + y \leq 5) \land (x + y \geq 5)$
		\item $x + y \neq 5 \equiv (x + y < 5) \lor (x + y > 5)$, however, we require additional variables to resolve the disjunction, $\lor$
	\end{itemize}

\end{frame}

\begin{frame}{Arithmetic constraints}
  Many other arithmetic operations exist:

  \begin{itemize}
    \item Absolute value $|a|$
    \item Modulo $x \% y$
    \item Minimum $min(x,y,z)$
    \item Maximum $max(x,y,z)$
  \end{itemize}

  $ $\\
  These can be used in arithmetic constraints too (\eg arithmetic operators + comparison).

  $ $\\
  Just like simple constraints, arithmetic constraints are Boolean-valued expressions too; and can be used in Boolean-valued expressions.

  $ $\\
  A \defined{nested expression} nests all sorts of Boolean operators and/or arithmetic constraints and operators. A contrived example: $(a \vee (|x - y| > z/2)) \rightarrow (r * s \neq  max(x, y-t, |z-3|)) \wedge (b \otimes (c \leftrightarrow  d))$

\end{frame}


\subsection{Global constraints}

\begin{frame}{Global constraints}

A \defined{global constraint} is a named constraint that captures a non-trivial relationship between decision variables (common combinatorial substructure).
Often, there is no equivalent operation in standard mathematics, though they frequently appear in combinatorial problems. \Eg:

\parBreak

  \begin{itemize}
	  \item $\cons{AllDifferent}{x, y, z}$
	  \item $\cons{Circuit}{\sequence{x,y,z}}$
	  \item $\cons{Table}{\sequence{x,y}, \sequence{\sequence{1,2},\sequence{1,4},\sequence{3,4}}}$
	    \item \ldots
  \end{itemize}

  \vfill
	\centering
	The globals are the subject of the next lecture

\end{frame}


\subsection{Objective functions}

\begin{frame}{Objective functions}

	The \defined{objective function} is an expression that we aim to minimize or maximize (\eg the sum of must-try factors).
	The objective does not remove solutions, but instead \stressed{ranks} them.

	$$
	x \in \interval{1}{5}, x \leq 3, \text{maximize}~x
	$$

	 This model still has 3 solutions, but one is \defined{optimal}.

  % $ $\\
  % \textit{Global functions} are valid integer-valued expressions too, as are nested arithmetic expressions (in high-level languages at least).

	 \parBreak

	The objective can be any expression, including global functions or other constraints.
  % Sometimes, we want to relax a constraint by allowing it to be violated.
	A \defined{hard constraint} can be modeled as a \defined{soft constraint} through the objective.

  % When we add a constraint (Boolean-valued expression) to the objective function, we call that constraint a \defined{soft constraint}, \eg $\Iverson{z = 0}$ below:

	\parBreak
	\Eg changing the above model, we prefer a solution where $x$ is even, but this may be violated. \textit{How many solutions and (optimal) solutions are there now?}
  \begin{align}
	x \in \interval{1}{5}, x \leq 3, \text{maximize}~x + \Iverson{x \% 2 = 0}
  \end{align}
\end{frame}

\begin{frame}{Focus point: reification}
	\begin{definition}[Reification]
		\defined{Reification} replaces a constraint $C$ (or any other Boolean-valued expression) for a Boolean \defined{reification} variable $b \in \set{0,1}$, and adds the constraint $b \iff C$ to the model.
		% $$
		%  C \equiv (b \iff (C \land b)) \land b
		% $$
	\end{definition}

	% \begin{figure}
	% 	\begin{forest}
	% 	for tree={draw, circle, minimum size=7mm, inner sep=1pt}
	% 	[{$\implies$}
	% 		[{$\bvA$}]
	% 		[{$x < y$}]
	% 	]
	% \end{forest}
	% 	\begin{forest}
	% 	for tree={draw, circle, minimum size=7mm, inner sep=1pt}
	% 	[{$\implies$}
	% 		[{$\bvA$}]
	% 		[{$\bvB$}]
	% 	]
	% \end{forest}
	% 	\caption{An expression tree for $\bvA \implies \bvB$ containing \stressed{three} expressions}
	% \end{figure}

  \begin{example}
	  Earlier, we rewrote the normalized linear disequality $x + y \neq 5$ to a disjunction. We can rewrite the disjunction by introducing reification variables $\bvA, \bvB \in \set{0,1}$:
  $$
  (x + y < 5) \lor (x + y > 5) \equiv \pause (\bvA \lor \bvB) \pause \land (\bvA \iff (x + y < 5)) \pause \land (\bvB \iff (x + y > 5))
  $$
	  One step closer to a constraint which is supported by the solver!
  	
  \end{example}

	Reification enables high-level modelling (\eg nested expressions), and efficient reuse of Boolean-valued expressions through their reification variable

\begin{frame}{The contract between Modeler and Solver}

	\begin{itemize}
		\item Each solver works in its own way, but  CPMpy allows you to use all solvers in \demph{(mostly)} the same way
		\item Given a model, the solver will \demph{(eventually)} return a solution if it exists
			\begin{itemize}
				\item There is no guarantee on \textit{which} solution it finds (first)
			\end{itemize}
		\item Solvers can find multiple solutions, but there is no guarantee on the order
			\begin{itemize}
				\item In the case of optimization, most solvers will only output \defined{intermediate} solutions which improve the objective
			\end{itemize}
		\item Most (but not all) solvers use \defined{exact} algorithms: they can find \stressed{all} solutions, or the \stressed{optimal} solution 
			\begin{itemize}
				\item Consequently, a solver can conclude a problem is \defined{unsatisfiable} (\ie has no solutions)
				\item The search space is \important{huge}, \eg $d^n$ for $n$ variables with domains of size $d$
				\item Still, the solver apparently \important{exhaustively explores} the full search space!
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Debugging}
	General advise for debugging when modeling from expert modeller\textbf{ Håkan Kjellerstrand}:
	\vfill
	
	\begin{itemize}
		\item Test the model \textbf{early and often}. This makes it easier to detect problems in the model. \vfill
		
		\item When a model is not working, \textbf{activate the constraints one by one} (\eg comment out the other constraints) to test which constraint is the culprit: 
		\vspace{-1.5em}
		\begin{algorithmic}
			\FOR{each constraint $c$ in \textit{Constraints}}
			\STATE print ``Trying:'', $c$
			\STATE \textit{Solve} $c$
			\ENDFOR
		\end{algorithmic}
		\vfill
		
		\item \textbf{Check the domains} (see lower). The domains should be as small as possible, but not smaller. If they are too large it can take a lot of time to get a solution. If they are too small, then there will be no solution. \vfill
	\end{itemize}
\end{frame}

	\begin{enumerate}

		\item the modeling library (CPMpy)
		\item the solver
	\end{enumerate}

\begin{frame}{Bug in the solver}
	You try with the default solver (or another one) and you get an error, or not the desired solution.
	\vfill
	
	\large
	Use a different solver and observe:
	\normalsize 
	\vfill
	
	\begin{enumerate}
		\item Outcome changes! It was a (rare) solver bug. Report it to the bug tracker of the modeling library or directly to the solver developers!
		\vfill
		
		\item Outcome is the same! Not a solver error after all \dots
	\end{enumerate}
\end{frame}

\begin{flashcardcpmpy}  % XXX does not have a preceding general slide...
	\begin{frame}{Debugging a modeling error -- CPMpy}
		You get an error when you create an expression?  \\ Quirks in Python/CPMpy (from last lecture):
		\vfill
		
		\begin{enumerate}
			\item \textbf{Logical and/or}:
			Use $\&$ and $|$, and make sure to always put the subexpressions in brackets. 
			\vfill
			
			\begin{example}
				write \cpminline{(x == 1) & (y == 0)} instead of \cpminline{x == 1 & y == 0}. The latter won't work.
				Python will think you meant \cpminline{x == (1 & y) == 0}.
			\end{example}
			\vfill
			
			\item you can write \cpminline{vars_list[other_var]} but you can’t write \cpminline{non_var_list[a_var]}. That is because the vars list knows CPMpy, and the \cpminline{non_var_list} does not. Wrap it: \cpminline{non_var_list = cp.cpm_array(non_var_list)} first.
			\vfill
			
			\item CPMpy overloads all/any/max/min/sum/abs to create expressions with them. Always use \cpminline{cp.sum(v)} instead of \cpminline{sum(v)}. You can also use directly \NumPy's \cpminline{v.sum()} instead, if \cpminline{v} is a matrix or tensor.
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Debugging a modeling error -- CPMpy}
		You get an error when you create an expression... but you do not know why!
		\vfill
		
		Print the constraints you create (or the sub-expressions), and check that the output matches what you wish to express!
		\vfill
		
		\begin{example}
			The following: \lstinputlisting[language=cpmpy,basicstyle=\footnotesize,firstline=3,lastline=5]{models_cpmpy/t3_sum.py}    
			will print \cpminline{[(IV0) + (IV2) (IV1) + (IV3)]}
			and you can see that it is not really a sum, but a list!
			
			Solution: Use \cpminline{cp.sum(x)} instead!
		\end{example}
		
	\end{frame}
